function devCommand(...args) {
	const root = clutchRoot();
	if (!root) return;
	const [binary, ...prefix] = command(root);
	const argv = [...prefix, ...args];
	const task = new vscode.Task({ type: 'clutch', args }, vscode.workspace.getWorkspaceFolder(vscode.Uri.file(root)), args.join(' '), 'clutch', new vscode.ProcessExecution(binary, argv, { cwd: root }));
	vscode.tasks.executeTask(task);
}
